        <label for="transactions">Enter transactions (e.g., 0 1 100):</label>
        <textarea id="transactions" placeholder="Format: from to amount, one per line"></textarea>
        <button onclick="minimizeCashFlow()">Minimize Cash Flow</button>
        <h2>Optimized Transactions:</h2>
        <div id="output"></div>
    </div>

    <script>
        function minimizeCashFlow() {
            const input = document.getElementById("transactions").value.trim().split("\n");
            const n = 100;
            let net = Array(n).fill(0);

            input.forEach(line => {
                const [from, to, amount] = line.trim().split(" ");
                if (from && to && amount) {
                    net[from] -= parseInt(amount);
                    net[to] += parseInt(amount);
                }
            });

            const result = [];
            function getMax(arr) {
                return arr.reduce((a, b, i) => b > arr[a] ? i : a, 0);
            }
            function getMin(arr) {
                return arr.reduce((a, b, i) => b < arr[a] ? i : a, 0);
            }

            function settle() {
                const mx = getMax(net), mn = getMin(net);
                if (net[mx] === 0 && net[mn] === 0) return;
                const minVal = Math.min(net[mx], -net[mn]);
                net[mx] -= minVal;
                net[mn] += minVal;
                result.push(`Person ${mn} pays ${minVal} to Person ${mx}`);
                settle();
            }

            settle();
            document.getElementById("output").innerHTML = result.join("<br>");
        }
    </script>
